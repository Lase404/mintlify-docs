---
title: "POST /api/quote"
description: "Create a new payment quote and deposit address."
---

<ParamField body="fromChainId" type="number" required>
  Source chain ID (e.g. `8453` for Base, `1` for Ethereum, `56` for BSC)
</ParamField>

<ParamField body="toChainId" type="number" required>
  Destination chain ID where the recipient receives funds
</ParamField>

<ParamField body="fromToken" type="string" required>
  Token contract address on the source chain. Use `0x0000000000000000000000000000000000000000` for the native token.
</ParamField>

<ParamField body="toToken" type="string" required>
  Token contract address on the destination chain. Use `0x0000000000000000000000000000000000000000` for the native token.
</ParamField>

<ParamField body="amount" type="string" required>
  Amount to send in human-readable units (e.g. `"1"` for 1 ETH)
</ParamField>

<ParamField body="recipient" type="string" required>
  Wallet address that will receive the funds on the destination chain
</ParamField>

## Example request

```bash
curl -X POST http://159.223.110.114:3000/api/quote \
  -H "Authorization: Bearer YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "fromChainId": 8453,
    "toChainId": 56,
    "fromToken": "0x0000000000000000000000000000000000000000",
    "toToken": "0x0000000000000000000000000000000000000000",
    "amount": "1",
    "recipient": "0xa110c77fa4b07ab601e63edEAd99ddb8f1df22c"
  }'
```

## Example response

<ResponseField name="id" type="string">
  Unique payment ID — use this to poll status
</ResponseField>

<ResponseField name="status" type="string">
  Always `pending` on creation
</ResponseField>

<ResponseField name="depositAddress" type="string">
  Address the user must send funds to
</ResponseField>

<ResponseField name="fromChainId" type="number">
  Source chain ID
</ResponseField>

<ResponseField name="toChainId" type="number">
  Destination chain ID
</ResponseField>

<ResponseField name="fromToken" type="string">
  Token address on source chain
</ResponseField>

<ResponseField name="toToken" type="string">
  Token address on destination chain
</ResponseField>

<ResponseField name="amount" type="string">
  Amount expected
</ResponseField>

<ResponseField name="recipient" type="string">
  Recipient wallet address
</ResponseField>

<ResponseField name="expiresAt" type="string">
  ISO timestamp — payment must arrive before this
</ResponseField>
